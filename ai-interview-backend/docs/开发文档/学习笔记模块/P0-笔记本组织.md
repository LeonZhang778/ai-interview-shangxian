## 1. 背景与目标

### 当前问题
- 笔记平铺列表，没有层级结构
- 笔记多了（>50 篇）查找困难
- 缺少分类管理能力

### 目标
- 新增 `notebooks` 表，支持多级文件夹树
- 每篇笔记归属一个笔记本
- 左侧边栏显示笔记本树，可展开/折叠
- 支持拖拽笔记到不同笔记本
- 笔记本支持：创建/重命名/删除/排序

from sqlalchemy.orm import relationship
from .base import BaseModel


class Notebook(BaseModel):
    __tablename__ = "notebooks"

    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    user_id = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False, index=True)
    name = Column(String(100), nullable=False)
    parent_id = Column(Integer, ForeignKey("notebooks.id", ondelete="CASCADE"), nullable=True, index=True)
    sort_order = Column(Integer, default=0)  # 同级排序
    icon = Column(String(20), nullable=True)  # 可选图标 emoji

    # 自引用关系
    children = relationship("Notebook", backref="parent", remote_side=[id],
                            order_by="Notebook.sort_order", lazy="selectin")
    user = relationship("User", backref="notebooks")

from typing import Optional
from fastapi import APIRouter, Depends
from pydantic import BaseModel
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.client.deps import get_current_user
from app.db.session import get_db
from app.models.user import User
from app.schemas.response import ApiResponse

router = APIRouter()


class NotebookCreateBody(BaseModel):
    name: str
    parent_id: Optional[int] = None
    icon: Optional[str] = None


class NotebookUpdateBody(BaseModel):
    name: Optional[str] = None
    icon: Optional[str] = None
    sort_order: Optional[int] = None


@router.get("")
async def get_notebooks(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """获取用户的笔记本树（扁平列表+父子关系）"""
    from app.services.client.notebook_service import notebook_service
    result = await notebook_service.get_tree(db=db, user_id=current_user.id)
    return ApiResponse.success(data=result)


@router.post("")
async def create_notebook(
    body: NotebookCreateBody,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    from app.services.client.notebook_service import notebook_service
    result = await notebook_service.create(
        db=db, user_id=current_user.id,
        name=body.name, parent_id=body.parent_id, icon=body.icon
    )
    return ApiResponse.success(data=result)


@router.put("/{notebook_id}")
async def update_notebook(
    notebook_id: int,
    body: NotebookUpdateBody,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    from app.services.client.notebook_service import notebook_service
    result = await notebook_service.update(
        db=db, user_id=current_user.id, notebook_id=notebook_id,
        **body.model_dump(exclude_unset=True)
    )
    return ApiResponse.success(data=result)


@router.delete("/{notebook_id}")
async def delete_notebook(
    notebook_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):

    result = await notebook_service.delete(db=db, user_id=current_user.id, notebook_id=notebook_id)
    return ApiResponse.success(message=result["message"])

### 3.2 Notebook Service

**文件：`app/services/client/notebook_service.py`**

```python
import logging
from typing import Optional, List, Dict
from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.notebook import Notebook
from app.models.learning_note import LearningNote
from app.exceptions.http_exceptions import NotFoundError, ValidationError

logger = logging.getLogger(__name__)


class NotebookService:

    @staticmethod
    async def get_tree(db: AsyncSession, user_id: int) -> Dict:
        """获取笔记本树，返回嵌套结构"""
        query = select(Notebook).where(
            Notebook.user_id == user_id
        ).order_by(Notebook.sort_order, Notebook.id)
        result = await db.execute(query)
        notebooks = result.scalars().all()

        # 构建树
        notebook_map = {n.id: {
            "id": n.id,
            "name": n.name,
            "parent_id": n.parent_id,
            "sort_order": n.sort_order,
            "icon": n.icon,
            "children": [],
            "note_count": 0,
        } for n in notebooks}

        # 统计每个笔记本的笔记数
        count_query = select(
            LearningNote.notebook_id,
            func.count(LearningNote.id)
        ).where(
            LearningNote.user_id == user_id,
            LearningNote.notebook_id.isnot(None)
        ).group_by(LearningNote.notebook_id)
        count_result = await db.execute(count_query)
        for nb_id, count in count_result:
            if nb_id in notebook_map:
                notebook_map[nb_id]["note_count"] = count

        # 组装树
        tree = []
        for n_id, item in notebook_map.items():
            if item["parent_id"] and item["parent_id"] in notebook_map:
                notebook_map[item["parent_id"]]["children"].append(item)
            else:
                tree.append(item)

        return {"tree": tree, "flat": list(notebook_map.values())}

    @staticmethod
    async def create(db: AsyncSession, user_id: int, name: str,
                     parent_id: Optional[int] = None, icon: Optional[str] = None) -> Dict:
        notebook = Notebook(
            user_id=user_id, name=name.strip(),
            parent_id=parent_id, icon=icon
        )
        db.add(notebook)
        await db.commit()
        await db.refresh(notebook)
        return {"id": notebook.id, "name": notebook.name}

    @staticmethod
    async def update(db: AsyncSession, user_id: int, notebook_id: int, **kwargs) -> Dict:
        notebook = await db.get(Notebook, notebook_id)
        if not notebook or notebook.user_id != user_id:
            raise NotFoundError(message="笔记本不存在")
        for key, value in kwargs.items():
            if value is not None and hasattr(notebook, key):
                setattr(notebook, key, value)
        await db.commit()
        return {"id": notebook.id, "message": "更新成功"}

    @staticmethod
    async def delete(db: AsyncSession, user_id: int, notebook_id: int) -> Dict:
        notebook = await db.get(Notebook, notebook_id)
        if not notebook or notebook.user_id != user_id:
            raise NotFoundError(message="笔记本不存在")
        # 将子笔记本和笔记移动到父笔记本
        sub_query = select(Notebook).where(Notebook.parent_id == notebook_id)
        sub_result = await db.execute(sub_query)
        for sub in sub_result.scalars():
            sub.parent_id = notebook.parent_id

        note_query = select(LearningNote).where(LearningNote.notebook_id == notebook_id)
        note_result = await db.execute(note_query)
        for note in note_result.scalars():
            note.notebook_id = notebook.parent_id

        await db.delete(notebook)
        await db.commit()
        return {"message": "笔记本已删除"}


notebook_service = NotebookService()
```
