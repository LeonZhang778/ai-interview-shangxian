# P2 · 导出功能

> 优先级：P2 | 预估工时：1-1.5 天 | 涉及端：后端 + 前端

---

## 1. 背景与目标

### 当前问题
- 笔记只能在站内查看，无法带到线下
- 用户可能想备份笔记或在其他工具（Typora、Obsidian、Notion）中编辑

### 目标
- 支持单篇笔记导出为 Markdown 文件
- 支持批量导出为 ZIP 压缩包
- 导出内容包含元数据（标题、标签、创建时间）

## 2. 技术方案

### 2.1 后端导出服务

**文件：`app/services/client/note_export_service.py`**（新增）

```python
import io
import zipfile
import logging
from datetime import datetime
from typing import List, Optional
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select

from app.models.learning_note import LearningNote

logger = logging.getLogger(__name__)


class NoteExportService:
    """笔记导出服务"""

    def __init__(self, db: AsyncSession):
        self.db = db

    async def export_single(self, note_id: int, user_id: int) -> Optional[io.BytesIO]:
        """导出单篇笔记为 Markdown"""
        result = await self.db.execute(
            select(LearningNote).where(
                LearningNote.id == note_id,
                LearningNote.user_id == user_id,
            )
        )
        note = result.scalar_one_or_none()
        if not note:
            return None

        content = self._build_markdown(note)
        buf = io.BytesIO(content.encode("utf-8"))
        return buf

    async def export_batch(
        self, user_id: int, note_ids: Optional[List[int]] = None
    ) -> io.BytesIO:
        """批量导出笔记为 ZIP 压缩包"""
        query = select(LearningNote).where(LearningNote.user_id == user_id)
        if note_ids:
            query = query.where(LearningNote.id.in_(note_ids))
        result = await self.db.execute(query)
        notes = result.scalars().all()

        buf = io.BytesIO()
        with zipfile.ZipFile(buf, "w", zipfile.ZIP_DEFLATED) as zf:
            for note in notes:
                content = self._build_markdown(note)
                safe_title = self._safe_filename(note.title)
                zf.writestr(f"notes/{safe_title}.md", content.encode("utf-8"))

        buf.seek(0)
        return buf

    def _build_markdown(self, note: LearningNote) -> str:
        """将笔记转为 Markdown 字符串（含元数据 front-matter）"""
        created = note.created_at.strftime("%Y-%m-%d %H:%M:%S")
        updated = note.updated_at.strftime("%Y-%m-%d %H:%M:%S")
        tags = note.tags or []

        lines = [
            "---",
            f"title: \"{note.title}\"",
            f"type: {note.note_type or 'note'}",
            f"created_at: {created}",
            f"updated_at: {updated}",
            f"tags: [{', '.join(tags)}]",
            "---",
            "",
            f"# {note.title}",
            "",
            note.content,
        ]
        return "\n".join(lines)

    @staticmethod
    def _safe_filename(title: str) -> str:
        """过滤文件名中的非法字符"""
        invalid = r'<>:"/\\|?*'
        for ch in invalid:
            title = title.replace(ch, "_")
        return title[:200]
```

### 2.2 后端 API

**文件：`app/api/client/v1/note_export.py`**（新增）

```python
import logging
from fastapi import APIRouter, Depends, Query
from fastapi.responses import StreamingResponse
from sqlalchemy.ext.asyncio import AsyncSession

from app.db.session import get_db
from app.api.client.deps import get_current_user
from app.models.user import User
from app.services.client.note_export_service import NoteExportService

router = APIRouter()
logger = logging.getLogger(__name__)


@router.get("/export/{note_id}")
async def export_single_note(
    note_id: int,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """导出单篇笔记为 Markdown"""
    service = NoteExportService(db)
    buf = await service.export_single(note_id, user.id)
    if not buf:
        from fastapi import HTTPException
        raise HTTPException(status_code=404, detail="笔记不存在")

    return StreamingResponse(
        buf,
        media_type="text/markdown",
        headers={
            "Content-Disposition": f'attachment; filename="note_{note_id}.md"',
        },
    )


@router.get("/export")
async def export_batch_notes(
    note_ids: str = Query("", description="逗号分隔的笔记 ID，为空则导出全部"),
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """批量导出笔记为 ZIP"""
    ids = [int(x.strip()) for x in note_ids.split(",") if x.strip()] if note_ids else None
    service = NoteExportService(db)
    buf = await service.export_batch(user.id, ids)

    return StreamingResponse(
        buf,
        media_type="application/zip",
        headers={
            "Content-Disposition": f'attachment; filename="notes_export_{user.id}.zip"',
        },
    )
```

## 3. 前端实现

### 3.1 单篇导出

```vue
<!-- 在笔记详情页添加导出按钮 -->
<button class="btn-export" @click="handleExportSingle">
  <span class="icon">📥</span> 导出 Markdown
</button>

<script setup>
import { downloadExport } from '../api/learningNote'

async function handleExportSingle() {
  try {
    const blob = await downloadExport(props.noteId)
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `note_${props.noteId}.md`
    a.click()
    URL.revokeObjectURL(url)
  } catch (e) {
    console.error('导出失败', e)
  }
}
</script>
```

### 3.2 批量导出

```vue
<!-- 在笔记列表页批量操作栏 -->
<template>
  <div class="batch-actions">
    <button @click="toggleSelectAll" :disabled="notes.length === 0">
      {{ allSelected ? '取消全选' : '全选' }}
    </button>
    <button
      @click="handleBatchExport"
      :disabled="selectedIds.length === 0"
      class="btn-export-batch"
    >
      📦 导出选中 ({{ selectedIds.length }})
    </button>
  </div>
</template>

<script setup>
async function handleBatchExport() {
  try {
    const blob = await downloadBatchExport(selectedIds.value)
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = 'notes_export.zip'
    a.click()
    URL.revokeObjectURL(url)
  } catch (e) {
    console.error('批量导出失败', e)
  }
}
</script>
```

### 3.3 新增 API 封装

```javascript
// 在 ai-interview-frontend/src/api/learningNote.js 中
export function downloadExport(noteId) {
  return api.get(`/note-export/export/${noteId}`, {
    responseType: 'blob',
  })
}

export function downloadBatchExport(noteIds) {
  return api.get(`/note-export/export`, {
    params: { note_ids: noteIds.join(',') },
    responseType: 'blob',
  })
}
```
