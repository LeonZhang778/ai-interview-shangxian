## 1. 背景与目标

### 当前问题
- `get_my_notes` 仅支持 `title` 字段的 `ilike` 模糊搜索
- `get_public_notes` 虽然支持 `title` / `content` / `tags` 的 `ilike` 搜索，但：
  - `ilike` 不走索引，大表性能差
  - 不支持中文分词
  - 不支持相关性排序
  - 不支持高亮片段

### 目标
- 基于 PostgreSQL 原生 `tsvector` 全文检索
- 支持中文分词（`zhparser` 或 `pg_trgm`）
- 支持标题 + 正文 + 标签联合搜索
- 返回相关性排序 + 匹配片段高亮
- 搜索性能稳定（百万级笔记 < 50ms）

---

## 2. 技术选型

### 选择 PostgreSQL 原生全文检索（不引入 Elasticsearch）

理由：
- 项目已使用 PostgreSQL，零额外运维成本
- 笔记量级短期内不会超过百万级，PG 全文检索足够
- 避免引入 ES 的运维复杂度

### 中文分词方案

**推荐方案：`pg_trgm` 扩展（无需额外安装）**

```sql
CREATE EXTENSION IF NOT EXISTS pg_trgm;
```

`pg_trgm` 基于三元组匹配，对中文也有一定的支持，且是 PostgreSQL 自带扩展。
如果项目需要更精细的中文分词，可考虑安装 `zhparser`（基于 SCWS），但需要编译安装。

**替代方案：后端分词 + tsvector**

```python
# 在后端用 jieba 分词后拼接 tsvector 存储
import jieba
words = " ".join(jieba.cut(text))
# 存储 words 到 tsvector 列
```

```python
Revises: f1a2b3c4d5e6

from alembic import op
import sqlalchemy as sa
from sqlalchemy.dialects.postgresql import TSVECTOR


def upgrade() -> None:
    # 启用 pg_trgm 扩展
    op.execute("CREATE EXTENSION IF NOT EXISTS pg_trgm")

    # 新增 search_vector 列
    op.add_column('learning_notes',
        sa.Column('search_vector', TSVECTOR(), nullable=True)
    )

    # 创建 GIN 索引
    op.create_index('ix_learning_notes_search_vector',
        'learning_notes', ['search_vector'],
        postgresql_using='gin'
    )

    # 创建更新 search_vector 的触发器函数
    op.execute("""
    CREATE OR REPLACE FUNCTION learning_notes_search_update()
    RETURNS trigger AS $$
    BEGIN
        NEW.search_vector = to_tsvector('simple',
            COALESCE(NEW.title, '') || ' ' ||
            COALESCE(NEW.content, '') || ' ' ||
            COALESCE(NEW.tags, '')
        );
        RETURN NEW;
    END;
    $$ LANGUAGE plpgsql;
    """)

    # 创建触发器，insert/update 时自动更新 search_vector
    op.execute("""
    CREATE TRIGGER trg_learning_notes_search
    BEFORE INSERT OR UPDATE OF title, content, tags
    ON learning_notes
    FOR EACH ROW
    EXECUTE FUNCTION learning_notes_search_update();
    """)

    # 为已有数据初始化 search_vector
    op.execute("""
    UPDATE learning_notes SET search_vector = to_tsvector('simple',
        COALESCE(title, '') || ' ' ||
        COALESCE(content, '') || ' ' ||
        COALESCE(tags, '')
    );
    """)


def downgrade() -> None:
    op.execute("DROP TRIGGER IF EXISTS trg_learning_notes_search ON learning_notes")
    op.execute("DROP FUNCTION IF EXISTS learning_notes_search_update()")
    op.drop_index('ix_learning_notes_search_vector', table_name='learning_notes')
    op.drop_column('learning_notes', 'search_vector')
    op.execute("DROP EXTENSION IF EXISTS pg_trgm")
```

## 4. 后端实现

### 4.1 修改搜索接口

**文件：`app/services/client/learning_note_service.py`**

```python
from sqlalchemy import func, text
from sqlalchemy.dialects.postgresql import TSVECTOR, REGCONFIG

@staticmethod
async def search_notes(
    db: AsyncSession,
    user_id: int,
    keyword: str,
    note_type: Optional[str] = None,
    page: int = 1,
    size: int = 20,
    scope: str = "my"  # my / public
) -> Dict:
    """
    全文搜索笔记（使用 PostgreSQL tsvector）

    Args:
        keyword: 搜索关键词
        scope: "my" 搜索我的笔记，"public" 搜索公开广场
    """
    # 使用 plainto_tsquery 将用户输入转成查询词
    query_vec = func.plainto_tsquery('simple', keyword)

    conditions = [
        LearningNote.search_vector.op("@@")(query_vec)  # @@ 是 tsquery 匹配操作符
    ]

    if scope == "my":
        conditions.append(LearningNote.user_id == user_id)
    else:
        conditions.append(LearningNote.visibility == "public")
        conditions.append(LearningNote.status == "published")

    if note_type:
        conditions.append(LearningNote.note_type == note_type)

    # 相关性排序：ts_rank 根据词频和位置打分
    rank = func.ts_rank(LearningNote.search_vector, query_vec)

    # 生成高亮片段
    headline = func.ts_headline(
        'simple',
        LearningNote.content,
        query_vec,
        'MaxWords=30, MinWords=15, StartSel=<mark>, StopSel=</mark>'
    )

    query = select(
        LearningNote,
        rank.label("relevance"),
        headline.label("headline")
    ).where(*conditions).order_by(rank.desc())

    # 分页
    total_query = select(func.count()).select_from(query.subquery())
    total = await db.scalar(total_query)

    offset = (page - 1) * size
    result = await db.execute(query.offset(offset).limit(size))
    rows = result.all()

    items = []
    for row in rows:
        note = row[0]  # LearningNote 对象
        items.append({
            "id": note.id,
            "title": note.title,
            "note_type": note.note_type,
            "tags": note.tags,
            "view_count": note.view_count,
            "like_count": note.like_count,
            "relevance": float(row.relevance) if row.relevance else 0,
            "headline": row.headline or "",
            "created_at": note.created_at.isoformat() if note.created_at else None,
        })

    from math import ceil
    return {
        "items": items,
        "total": total,
        "per_page": size,
        "current_page": page,
        "last_page": ceil(total / size) if size > 0 else 0,
        "has_more": page < ceil(total / size) if size > 0 and total > 0 else False,
    }
```

### 4.2 新增搜索 API

**文件：`app/api/client/v1/learning_note.py`**

```python
@router.get("/search")
async def search_notes(
    keyword: str = Query(..., min_length=1, max_length=100),
    note_type: Optional[str] = Query(None),
    page: int = Query(1, ge=1),
    size: int = Query(20, ge=1, le=50),
    scope: str = Query("my"),
    current_user: User | None = Depends(get_current_user_optional),
    db: AsyncSession = Depends(get_db),
):
    """全文搜索笔记（支持 my 和 public 两种范围）"""
    if scope == "my" and not current_user:
        return ApiResponse.error(message="需要登录才能搜索我的笔记")
    user_id = current_user.id if current_user else None
    result = await learning_note_service.search_notes(
        db=db, user_id=user_id, keyword=keyword,
        note_type=note_type, page=page, size=size, scope=scope
    )
    return ApiResponse.success(data=result)
```

## 8. 性能说明

### 索引大小估算
- 10 万篇笔记，`search_vector` GIN 索引约 200-500MB
- `pg_trgm` 索引额外 < 100MB

### 查询计划示例
```sql
EXPLAIN ANALYZE
SELECT id, title, ts_rank(search_vector, query) AS rank
FROM learning_notes, plainto_tsquery('simple', '面试') query
WHERE search_vector @@ query
  AND user_id = 1
ORDER BY rank DESC
LIMIT 20;
-- 预期：Bitmap Heap Scan + GIN Index Scan
-- 10万行数据下 < 10ms
```
