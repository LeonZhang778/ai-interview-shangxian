# P2 · 关联推荐

> 优先级：P2 | 预估工时：1.5 天 | 涉及端：后端

---

## 1. 背景与目标

### 当前问题
- 用户查看一篇笔记时，没有其他相关笔记的推荐
- 发现新笔记全靠搜索，缺少主动推荐
- 笔记之间缺乏关联，知识体系是孤立的

### 目标
- 笔记详情页展示"相关笔记"推荐（基于标签重合度）
- 面试结果页推荐相关笔记（基于知识点的关联）
- 实现基于标签的简单关联算法，无需引入外部推荐系统

---

## 2. 技术方案

### 2.1 关联算法

采用**基于标签的 Jaccard 相似度**，简单高效：

```
J(A, B) = |tags(A) ∩ tags(B)| / |tags(A) ∪ tags(B)|
```

- 仅对比当前用户和其他用户的公开笔记
- 如果标签为空，回退到笔记类型 + 标题关键词匹配
- 按相似度降序排列，取 Top 5

### 2.2 推荐服务

**文件：`app/services/client/note_recommend_service.py`**（新增）

```python
import logging
from typing import Dict, List, Optional
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func, desc, and_, or_

from app.models.learning_note import LearningNote

logger = logging.getLogger(__name__)


class NoteRecommendService:
    """笔记关联推荐服务"""

    def __init__(self, db: AsyncSession):
        self.db = db

    async def get_related_notes(
        self, note_id: int, user_id: int, limit: int = 5
    ) -> List[Dict]:
        """根据标签相似度推荐相关笔记"""
        # 1. 获取当前笔记
        result = await self.db.execute(
            select(LearningNote).where(
                LearningNote.id == note_id,
                LearningNote.is_deleted == False,
            )
        )
        note = result.scalar_one_or_none()
        if not note:
            return []

        tags = note.tags or []
        note_type = note.note_type or ""

        # 2. 候选集：公开笔记 + 用户自己的笔记（排除自身）
        candidates_query = select(LearningNote).where(
            LearningNote.id != note_id,
            LearningNote.is_deleted == False,
            or_(
                LearningNote.is_public == True,
                LearningNote.user_id == user_id,
            ),
        )
        # 如果笔记有标签，加入标签过滤减少候选集
        if tags:
            tag_filters = []
            for tag in tags:
                tag_filters.append(LearningNote.tags.any(tag))
            candidates_query = candidates_query.where(or_(*tag_filters))

        candidates_result = await self.db.execute(candidates_query)
        candidates = candidates_result.scalars().all()

        # 3. 计算相似度并排序
        scored = []
        for candidate in candidates:
            c_tags = candidate.tags or []
            similarity = self._jaccard_similarity(tags, c_tags)

            # 类型加分（相同类型 +0.1）
            type_bonus = 0.1 if candidate.note_type == note_type else 0

            score = similarity + type_bonus
            if score > 0:
                scored.append((score, candidate))

        scored.sort(key=lambda x: -x[0])
        top = scored[:limit]

        # 4. 格式化结果
        return [
            {
                "id": candidate.id,
                "title": candidate.title,
                "note_type": candidate.note_type,
                "tags": candidate.tags or [],
                "summary": candidate.summary or "",
                "similarity": round(score, 3),
                "created_at": candidate.created_at.isoformat() if candidate.created_at else None,
            }
            for score, candidate in top
        ]

    async def get_recommend_for_interview(
        self, user_id: int, keywords: List[str], limit: int = 3
    ) -> List[Dict]:
        """面试结果页推荐相关笔记（基于关键词）"""
        if not keywords:
            return []

        query = select(LearningNote).where(
            LearningNote.user_id == user_id,
            LearningNote.is_deleted == False,
        )
        # 根据关键词匹配标题或标签
        keyword_filters = []
        for kw in keywords:
            keyword_filters.append(LearningNote.tags.any(kw))
            keyword_filters.append(LearningNote.title.ilike(f"%{kw}%"))
        query = query.where(or_(*keyword_filters))
        query = query.order_by(desc(LearningNote.updated_at)).limit(limit)

        result = await self.db.execute(query)
        notes = result.scalars().all()

        return [
            {
                "id": n.id,
                "title": n.title,
                "note_type": n.note_type,
                "tags": n.tags or [],
                "summary": n.summary or "",
                "created_at": n.created_at.isoformat() if n.created_at else None,
            }
            for n in notes
        ]

    @staticmethod
    def _jaccard_similarity(a: List[str], b: List[str]) -> float:
        """计算 Jaccard 相似度"""
        set_a = set(t.lower() for t in a if t)
        set_b = set(t.lower() for t in b if t)
        if not set_a or not set_b:
            return 0.0
        intersection = set_a & set_b
        union = set_a | set_b
        return len(intersection) / len(union)
```

### 2.3 后端 API

**在 `app/api/client/v1/public_note.py` 中追加**

```python
@router.get("/public/{note_id}/related")
async def get_related_notes(
    note_id: int,
    limit: int = Query(5, ge=1, le=20),
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """获取笔记的相关推荐"""
    from app.services.client.note_recommend_service import NoteRecommendService
    service = NoteRecommendService(db)
    notes = await service.get_related_notes(note_id, user.id, limit)
    return ApiResponse.success(data={"items": notes})


@router.post("/recommend/interview")
async def recommend_for_interview(
    body: dict,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """面试结果页推荐相关笔记"""
    from app.services.client.note_recommend_service import NoteRecommendService
    service = NoteRecommendService(db)
    notes = await service.get_recommend_for_interview(
        user.id, body.get("keywords", []),
    )
    return ApiResponse.success(data={"items": notes})
```

import { ref, onMounted } from 'vue'
import { getRelatedNotes } from '../api/learningNote'

const relatedNotes = ref([])

async function loadRelated() {
  try {
    const res = await getRelatedNotes(props.noteId)
    relatedNotes.value = res.items || []
  } catch (e) {
    console.error('加载相关推荐失败', e)
  }
}
