---

## 1. 背景与目标

### 当前问题
- 错题类型（`note_type="mistake"`）只是一个标签，没有系统性的聚合展示
- 用户不知道自己的薄弱知识点分布
- 无法从面试回流数据中自动统计薄弱知识点
- 缺少针对薄弱点的学习推荐

### 目标
- 错题聚合仪表盘：按知识点聚合错题频率
- 薄弱点热力图：从面试回流的报告数据自动统计
- 推荐针对性练习：基于薄弱点推荐面经或面试题
- 复用现有 `Analysis.vue` 的 ECharts 图表能力

## 2. 数据来源

### 2.1 错题数据来源

| 来源 | 数据内容 | 获取方式 |
|------|----------|----------|
| 面试报告 | `report.weaknesses` 薄弱点列表 | 面试完成时自动记录 |
| 面试评分 | 每题得分 < 6 分的题目 | `InterviewMessage.score` 字段 |
| 手动错题笔记 | `note_type="mistake"` 的笔记 | 已有条件筛选 |
| 手动错题标签 | 笔记中的 `tags` 字段 | 标签解析 |

```python
from .base import BaseModel


class WeaknessStat(BaseModel):
    """薄弱点统计（从面试报告自动聚合）"""
    __tablename__ = "weakness_stats"

    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    user_id = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False, index=True)
    keyword = Column(String(100), nullable=False)  # 薄弱知识点关键词
    count = Column(Integer, default=0)             # 出现次数
    avg_score = Column(Float, nullable=True)       # 平均得分
    last_occurred_at = Column(DateTime, nullable=True)  # 最近一次出现时间

    __table_args__ = (
        Index("ix_weakness_stats_user_keyword", "user_id", "keyword", unique=True),
    )
```

## 3. 后端实现

### 3.1 薄弱点统计服务

**文件：`app/services/client/weakness_service.py`**（新增）

```python
import json
import logging
from typing import Dict, List, Optional
from sqlalchemy import select, func, desc, update
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.weakness_stat import WeaknessStat
from app.models.learning_note import LearningNote
from app.models.interview import Interview
from app.models.interview_message import InterviewMessage
from datetime import datetime

logger = logging.getLogger(__name__)


class WeaknessService:

    @staticmethod
    async def record_from_interview(
        db: AsyncSession,
        user_id: int,
        interview_id: int
    ) -> None:
        """从面试报告记录薄弱点"""
        interview = await db.get(Interview, interview_id)
        if not interview or not interview.report:
            return

        try:
            report = json.loads(interview.report)
        except json.JSONDecodeError:
            return

        weaknesses = report.get("weaknesses", [])
        if not weaknesses:
            return

        now = datetime.utcnow()
        for kw in weaknesses:
            # 检查是否已存在
            query = select(WeaknessStat).where(
                WeaknessStat.user_id == user_id,
                WeaknessStat.keyword == kw.strip()
            )
            result = await db.execute(query)
            stat = result.scalar_one_or_none()

            if stat:
                stat.count += 1
                stat.last_occurred_at = now
            else:
                stat = WeaknessStat(
                    user_id=user_id,
                    keyword=kw.strip(),
                    count=1,
                    last_occurred_at=now,
                )
                db.add(stat)

        await db.commit()

    @staticmethod
    async def get_weakness_heatmap(
        db: AsyncSession,
        user_id: int
    ) -> Dict:
        """获取薄弱点热力图"""
        query = select(WeaknessStat).where(
            WeaknessStat.user_id == user_id
        ).order_by(desc(WeaknessStat.count)).limit(30)

        result = await db.execute(query)
        stats = result.scalars().all()

        items = [
            {
                "keyword": s.keyword,
                "count": s.count,
                "avg_score": s.avg_score,
                "last_occurred_at": s.last_occurred_at.isoformat() if s.last_occurred_at else None,
                # 热度分数：次数越多越热
                "heat": min(s.count / 10, 1.0),
            }
            for s in stats
        ]

        # 按次数排序
        items.sort(key=lambda x: x["count"], reverse=True)

        return {
            "items": items,
            "total": len(items),
            "top_3": items[:3] if items else [],
        }

    @staticmethod
    async def get_mistake_dashboard(
        db: AsyncSession,
        user_id: int
    ) -> Dict:
        """获取错题本仪表盘数据"""
        # 1. 错题笔记统计
        mistake_query = select(LearningNote).where(
            LearningNote.user_id == user_id,
            LearningNote.note_type == "mistake"
        ).order_by(desc(LearningNote.updated_at)).limit(10)
        mistake_result = await db.execute(mistake_query)
        mistakes = mistake_result.scalars().all()

        # 2. 薄弱点统计
        weakness_data = await WeaknessService.get_weakness_heatmap(db, user_id)

        return {
            "mistake_count": len(mistakes),
            "recent_mistakes": [
                {"id": n.id, "title": n.title, "tags": n.tags, "updated_at": n.updated_at.isoformat() if n.updated_at else None}
                for n in mistakes
            ],
            "weakness": weakness_data,
        }

    @staticmethod
    async def get_recommended_practice(
        db: AsyncSession,
        user_id: int,
        limit: int = 5
    ) -> List[Dict]:
        """基于薄弱点推荐练习内容"""
        # 获取 top 薄弱点
        weakness_query = select(WeaknessStat).where(
            WeaknessStat.user_id == user_id
        ).order_by(desc(WeaknessStat.count)).limit(limit)
        weakness_result = await db.execute(weakness_query)
        weaknesses = weakness_result.scalars().all()

        recommendations = []
        for w in weaknesses:
            # 查找与该关键词相关的面经
            from app.models.experience import Experience
            from sqlalchemy import or_

            exp_query = select(Experience).where(
                Experience.status == "approved",
                or_(
                    Experience.title.ilike(f"%{w.keyword}%"),
                    Experience.content.ilike(f"%{w.keyword}%"),
                    Experience.tags.ilike(f"%{w.keyword}%"),
                ),
            ).order_by(desc(Experience.like_count)).limit(3)

            exp_result = await db.execute(exp_query)
            exps = exp_result.scalars().all()

            recommendations.append({
                "keyword": w.keyword,
                "count": w.count,
                "experiences": [
                    {"id": e.id, "title": e.title, "company": e.company, "position": e.position}
                    for e in exps
                ],
            })

        return recommendations


weakness_service = WeaknessService()
```

```python
@router.get("/dashboard")
```
