## 1. 背景与目标

### 当前问题
- 笔记是纯私密的，优秀笔记无法被其他人看到
- 面经广场和笔记之间缺乏联动
- 缺乏社区氛围，用户之间不能互相学习

### 目标
- 允许用户将笔记设为公开（一键开关）
- 公开笔记展示在"社区笔记"板块
- 支持点赞、收藏、评论公开笔记
- 公开笔记作者可收到互动通知

from fastapi import APIRouter, Depends, Query
from sqlalchemy.ext.asyncio import AsyncSession

from app.db.session import get_db
from app.api.client.deps import get_current_user
from app.models.user import User
from app.schemas.response import ApiResponse

router = APIRouter()


@router.put("/{note_id}/publish")
async def toggle_public_note(
    note_id: int,
    body: dict,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """设置/取消公开"""
    service = NoteInteractionService(db)
    note = await service.update_public_status(note_id, user.id, body.get("is_public", False))
    if not note:
        return ApiResponse.error(message="笔记不存在")
    return ApiResponse.success(data={"note_id": note.id, "is_public": note.is_public})


@router.get("/public")
async def list_public_notes(
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=50),
    keyword: str = Query("", max_length=100),
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """浏览公开笔记"""
    service = NoteInteractionService(db)
    result = await service.list_public(page, page_size, keyword)
    return ApiResponse.success(data=result)


@router.get("/public/{note_id}")
async def get_public_note_detail(
    note_id: int,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """查看公开笔记详情"""
    service = NoteInteractionService(db)
    note = await service.get_public_detail(note_id, user.id)
    if not note:
        return ApiResponse.error(message="笔记不存在或未公开")
    return ApiResponse.success(data=note)


@router.post("/public/{note_id}/like")
async def like_public_note(
    note_id: int,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """点赞/取消点赞"""
    service = NoteInteractionService(db)
    result = await service.toggle_like(note_id, user.id)
    return ApiResponse.success(data=result)


@router.post("/public/{note_id}/collect")
async def collect_public_note(
    note_id: int,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """收藏/取消收藏"""
    service = NoteInteractionService(db)
    result = await service.toggle_collect(note_id, user.id)
    return ApiResponse.success(data=result)


@router.get("/public/{note_id}/comments")
async def get_comments(
    note_id: int,
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=50),
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """获取评论列表"""
    service = NoteInteractionService(db)
    result = await service.get_comments(note_id, page, page_size)
    return ApiResponse.success(data=result)


@router.post("/public/{note_id}/comments")
async def create_comment(
    note_id: int,
    body: dict,
    db: AsyncSession = Depends(get_db),
    user: User = Depends(get_current_user),
):
    """发表评论"""
    service = NoteInteractionService(db)
    comment = await service.create_comment(
        note_id, user.id, body.get("content", ""),
        parent_id=body.get("parent_id"),
    )
    return ApiResponse.success(data=comment)

### 3.2 互动服务

**文件：`app/services/client/note_interaction_service.py`**（新增）

```python
import logging
from datetime import datetime, timezone
from typing import Dict, Optional, List
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select, func, desc, and_, or_, update

from app.models.learning_note import LearningNote
from app.models.note_interaction import (
    PublicNoteLike, PublicNoteCollection, PublicNoteComment,
)
from app.models.notification import Notification

logger = logging.getLogger(__name__)


class NoteInteractionService:
    """笔记互动服务"""

    def __init__(self, db: AsyncSession):
        self.db = db

    async def update_public_status(self, note_id: int, user_id: int, is_public: bool) -> Optional[LearningNote]:
        """更新笔记公开状态"""
        result = await self.db.execute(
            select(LearningNote).where(
                LearningNote.id == note_id,
                LearningNote.user_id == user_id,
            )
        )
        note = result.scalar_one_or_none()
        if not note:
            return None
        note.is_public = is_public
        note.published_at = datetime.now(timezone.utc) if is_public else None
        await self.db.commit()
        await self.db.refresh(note)
        return note

    async def list_public(self, page: int, page_size: int, keyword: str = "") -> Dict:
        """浏览公开笔记列表"""
        query = select(LearningNote).where(
            LearningNote.is_public == True,
            LearningNote.is_deleted == False,
        )
        if keyword:
            query = query.where(
                or_(
                    LearningNote.title.ilike(f"%{keyword}%"),
                    LearningNote.content.ilike(f"%{keyword}%"),
                )
            )
        query = query.order_by(desc(LearningNote.published_at))

        # 总数
        count_query = select(func.count()).select_from(query.subquery())
        total = (await self.db.execute(count_query)).scalar() or 0

        # 分页
        query = query.offset((page - 1) * page_size).limit(page_size)
        result = await self.db.execute(query)
        notes = result.scalars().all()

        items = [
            {
                "id": n.id,
                "title": n.title,
                "note_type": n.note_type,
                "tags": n.tags,
                "summary": n.summary,
                "view_count": n.view_count,
                "like_count": n.like_count,
                "collect_count": n.collect_count,
                "comment_count": n.comment_count,
                "created_at": n.created_at.isoformat() if n.created_at else None,
                "author": {
                    "id": n.user.id,
                    "nickname": n.user.nickname,
                    "avatar": n.user.avatar_url,
                },
            }
            for n in notes
        ]
        return {"items": items, "total": total, "page": page, "page_size": page_size}

    async def get_public_detail(self, note_id: int, current_user_id: int) -> Optional[Dict]:
        """查看公开笔记详情"""
        result = await self.db.execute(
            select(LearningNote).where(
                LearningNote.id == note_id,
                LearningNote.is_public == True,
                LearningNote.is_deleted == False,
            )
        )
        note = result.scalar_one_or_none()
        if not note:
            return None

        # 增加阅读计数
        note.view_count = (note.view_count or 0) + 1
        await self.db.commit()

        # 查询当前用户互动状态
        liked = False
        collected = False
        if current_user_id:
            lr = await self.db.execute(
                select(PublicNoteLike).where(
                    PublicNoteLike.note_id == note_id,
                    PublicNoteLike.user_id == current_user_id,
                )
            )
            liked = lr.scalar_one_or_none() is not None
            cr = await self.db.execute(
                select(PublicNoteCollection).where(
                    PublicNoteCollection.note_id == note_id,
                    PublicNoteCollection.user_id == current_user_id,
                )
            )
            collected = cr.scalar_one_or_none() is not None

        return {
            "id": note.id,
            "title": note.title,
            "content": note.content,
            "note_type": note.note_type,
            "tags": note.tags,
            "summary": note.summary,
            "view_count": note.view_count,
            "like_count": note.like_count,
            "collect_count": note.collect_count,
            "comment_count": note.comment_count,
            "created_at": note.created_at.isoformat() if note.created_at else None,
            "author": {
                "id": note.user.id,
                "nickname": note.user.nickname,
                "avatar": note.user.avatar_url,
            },
            "interaction": {"liked": liked, "collected": collected},
        }

    async def toggle_like(self, note_id: int, user_id: int) -> Dict:
        """切换点赞状态"""
        result = await self.db.execute(
            select(PublicNoteLike).where(
                PublicNoteLike.note_id == note_id,
                PublicNoteLike.user_id == user_id,
            )
        )
        existing = result.scalar_one_or_none()
        if existing:
            await self.db.delete(existing)
            await self._update_count(note_id, "like_count", -1)
            liked = False
        else:
            self.db.add(PublicNoteLike(note_id=note_id, user_id=user_id))
            await self._update_count(note_id, "like_count", 1)
            liked = True
            # 创建通知
            await self._create_notification(note_id, user_id, "like")
        await self.db.commit()
        return {"liked": liked}

    async def toggle_collect(self, note_id: int, user_id: int) -> Dict:
        """切换收藏状态"""
        result = await self.db.execute(
            select(PublicNoteCollection).where(
                PublicNoteCollection.note_id == note_id,
                PublicNoteCollection.user_id == user_id,
            )
        )
        existing = result.scalar_one_or_none()
        if existing:
            await self.db.delete(existing)
            await self._update_count(note_id, "collect_count", -1)
            collected = False
        else:
            self.db.add(PublicNoteCollection(note_id=note_id, user_id=user_id))
            await self._update_count(note_id, "collect_count", 1)
            collected = True
        await self.db.commit()
        return {"collected": collected}

    async def _update_count(self, note_id: int, field: str, delta: int):
        """更新计数（原子操作）"""
        await self.db.execute(
            update(LearningNote)
            .where(LearningNote.id == note_id)
            .values(**{field: LearningNote.__table__.c[field] + delta})
        )

    async def _create_notification(self, note_id: int, actor_id: int, action: str):
        """创建互动通知"""
        note_result = await self.db.execute(
            select(LearningNote).where(LearningNote.id == note_id)
        )
        note = note_result.scalar_one_or_none()
        if not note or note.user_id == actor_id:
            return  # 不给自己发送通知
        action_label = "点赞了" if action == "like" else "收藏了"
        notif = Notification(
            user_id=note.user_id,
            type="note_interaction",
            title=f"有人{action_label}你的笔记",
            content=f"你的笔记《{note.title}》被{action_label}",
            related_id=note_id,
            is_read=False,
        )
        self.db.add(notif)

    # 评论相关方法省略，模式与面经评论类似
```

### 4.2 社区笔记板块

**新建文件：`ai-interview-frontend/src/views/PublicNotes.vue`**

```vue
<template>
  <div class="public-notes-page">
    <div class="page-header">
      <h1>📚 社区笔记</h1>
      <p>看看其他用户分享的面试笔记</p>
    </div>

    <div class="search-bar">
      <input
        v-model="keyword"
        placeholder="搜索笔记标题..."
        @keyup.enter="search"
      />
    </div>

    <div class="note-grid" v-if="!loading">
      <div
        v-for="note in notes"
        :key="note.id"
        class="note-card"
        @click="goDetail(note.id)"
      >
        <div class="card-header">
          <span class="type-badge" :class="note.note_type">
            {{ typeLabel(note.note_type) }}
          </span>
        </div>
        <h3>{{ note.title }}</h3>
        <p class="summary">{{ note.summary || '暂无摘要' }}</p>
        <div class="tags">
          <span v-for="tag in (note.tags || [])" :key="tag" class="tag">
            {{ tag }}
          </span>
        </div>
        <div class="card-footer">
          <div class="author">
            <img :src="note.author.avatar || defaultAvatar" class="avatar" />
            <span>{{ note.author.nickname }}</span>
          </div>
          <div class="stats">
            <span>👁 {{ note.view_count }}</span>
            <span>👍 {{ note.like_count }}</span>
            <span>💬 {{ note.comment_count }}</span>
          </div>
        </div>
      </div>
    </div>
    <div v-else class="loading">加载中...</div>
  </div>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { getPublicNotes } from '../api/learningNote'

const router = useRouter()
const notes = ref([])
const loading = ref(true)
const keyword = ref('')

async function fetchNotes() {
  loading.value = true
  try {
    const res = await getPublicNotes({ keyword: keyword.value })
    notes.value = res.items || []
  } catch (e) {
    console.error(e)
  } finally {
    loading.value = false
  }
}

function search() {
  fetchNotes()
}

function goDetail(id) {
  router.push(`/public-notes/${id}`)
}

function typeLabel(type) {
  const map = { review: '复盘', mistake: '错题', experience: '面经', note: '笔记' }
  return map[type] || '笔记'
}

onMounted(fetchNotes)
</script>
```
