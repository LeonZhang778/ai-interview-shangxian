# P1 · 闪卡与间隔重复（Spaced Repetition）

> 优先级：P1 | 预估工时：2-3 天 | 涉及端：后端 + 前端

---

## 1. 背景与目标

### 当前问题
- 用户做完笔记后很少回头复习
- 艾宾浩斯遗忘曲线：学习后 1 小时遗忘 50%，1 天遗忘 70%
- 没有机制帮助用户对抗遗忘

### 目标
- 从笔记内容自动生成问答闪卡（Q&A Flashcard）
- 基于 SM-2 间隔重复算法安排复习计划
- 每天推送需要复习的闪卡
- 复习进度追踪（已掌握 / 需复习 / 已遗忘）

class Flashcard(BaseModel):
    """闪卡"""

    id = Column(Integer, primary_key=True, index=True, autoincrement=True)
    user_id = Column(Integer, ForeignKey("users.id", ondelete="CASCADE"), nullable=False, index=True)

    interval = Column(Integer, default=0)            # 复习间隔（天）
    repetitions = Column(Integer, default=0)         # 连续正确次数
    next_review_at = Column(DateTime, nullable=False, default=datetime.utcnow)  # 下次复习时间
    last_reviewed_at = Column(DateTime, nullable=True)  # 上次复习时间
    last_quality = Column(Integer, nullable=True)    # 上次评分（0-5）

    user = relationship("User", backref="flashcards")
    note = relationship("LearningNote", backref="flashcards")

    __table_args__ = (
        Index("ix_flashcards_user_review", "user_id", "next_review_at"),

## 3. SM-2 算法实现

**文件：`app/services/client/flashcard_service.py`**

```python
import logging
import math
from datetime import datetime, timedelta, timezone
from typing import Dict, List, Optional, Tuple
from sqlalchemy import select, func, desc, and_
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.flashcard import Flashcard
from app.models.learning_note import LearningNote
from app.exceptions.http_exceptions import NotFoundError, ValidationError

logger = logging.getLogger(__name__)


class SM2Algorithm:
    """
    SM-2 间隔重复算法（SuperMemo 2）

    评分标准（quality）：
    5 - 完美回答，立即想起
    4 - 正确回答，有些犹豫
    3 - 正确回答，需要较多努力
    2 - 错误回答，但正确答案看起来熟悉
    1 - 错误回答，不记得正确答案
    0 - 完全不记得
    """

    @staticmethod
    def calculate(quality: int, ease_factor: float, interval: int, repetitions: int) -> Dict:
        """
        根据 SM-2 算法计算新的复习参数

        Args:
            quality: 用户评分 0-5
            ease_factor: 当前难度系数
            interval: 当前间隔（天）
            repetitions: 当前连续正确次数

        Returns:
            { ease_factor, interval, repetitions, next_review_at }
        """
        # 质量评分必须在 0-5 之间
        quality = max(0, min(5, quality))

        if quality < 3:
            # 回答错误，重置
            repetitions = 0
            interval = 1
        else:
            # 回答正确
            if repetitions == 0:
                interval = 1
            elif repetitions == 1:
                interval = 6
            else:
                interval = round(interval * ease_factor)

            repetitions += 1

        # 更新难度系数
        new_ease = ease_factor + (0.1 - (5 - quality) * (0.08 + (5 - quality) * 0.02))
        ease_factor = max(1.3, new_ease)  # 最低 1.3

        # 最大间隔 365 天
        interval = min(interval, 365)

        next_review = datetime.utcnow() + timedelta(days=interval)

        return {
            "ease_factor": round(ease_factor, 2),
            "interval": interval,
            "repetitions": repetitions,
            "next_review_at": next_review,
        }


class FlashcardService:

    @staticmethod
    async def get_due_cards(
        db: AsyncSession,
        user_id: int,
        limit: int = 20
    ) -> List[Dict]:
        """获取今天需要复习的闪卡"""
        now = datetime.utcnow()
        query = select(Flashcard).where(
            Flashcard.user_id == user_id,
            Flashcard.next_review_at <= now,
        ).order_by(Flashcard.next_review_at).limit(limit)

        result = await db.execute(query)
        cards = result.scalars().all()

        return [
            {
                "id": c.id,
                "question": c.question,
                "answer": c.answer,
                "note_id": c.note_id,
                "ease_factor": c.ease_factor,
                "interval": c.interval,
                "repetitions": c.repetitions,
                "last_quality": c.last_quality,
                "source": c.source,
            }
            for c in cards
        ]

    @staticmethod
    async def review_card(
        db: AsyncSession,
        user_id: int,
        card_id: int,
        quality: int
    ) -> Dict:
        """
        复习一张闪卡，应用 SM-2 算法更新参数

        Args:
            quality: 0-5 评分
        """
        card = await db.get(Flashcard, card_id)
        if not card or card.user_id != user_id:
            raise NotFoundError(message="闪卡不存在")

        if quality < 0 or quality > 5:
            raise ValidationError(message="评分必须在 0-5 之间")

        # 应用 SM-2 算法
        result = SM2Algorithm.calculate(
            quality=quality,
            ease_factor=card.ease_factor,
            interval=card.interval,
            repetitions=card.repetitions,
        )

        # 更新闪卡
        card.ease_factor = result["ease_factor"]
        card.interval = result["interval"]
        card.repetitions = result["repetitions"]
        card.next_review_at = result["next_review_at"]
        card.last_reviewed_at = datetime.utcnow()
        card.last_quality = quality

        await db.commit()

        return {
            "id": card.id,
            "quality": quality,
            "next_review_at": result["next_review_at"].isoformat(),
            "interval_days": result["interval"],
            "ease_factor": result["ease_factor"],
            "repetitions": result["repetitions"],
            "message": "太棒了，继续保持！" if quality >= 4 else (
                "加油，下次会更好！" if quality >= 3 else "别灰心，再复习一次！"
            ),
        }

    @staticmethod
    async def get_stats(
        db: AsyncSession,
        user_id: int
    ) -> Dict:
        """获取闪卡学习统计"""
        now = datetime.utcnow()

        # 总闪卡数
        total = await db.scalar(
            select(func.count()).select_from(
                select(Flashcard).where(Flashcard.user_id == user_id).subquery()
            )
        )

        # 待复习数
        due = await db.scalar(
            select(func.count()).select_from(
                select(Flashcard).where(
                    Flashcard.user_id == user_id,
                    Flashcard.next_review_at <= now,
                ).subquery()
            )
        )

        # 已掌握（interval >= 21 天）
        mastered = await db.scalar(
            select(func.count()).select_from(
                select(Flashcard).where(
                    Flashcard.user_id == user_id,
                    Flashcard.interval >= 21,
                ).subquery()
            )
        )

        # 今日复习数
        today_start = now.replace(hour=0, minute=0, second=0, microsecond=0)
        reviewed_today = await db.scalar(
            select(func.count()).select_from(
                select(Flashcard).where(
                    Flashcard.user_id == user_id,
                    Flashcard.last_reviewed_at >= today_start,
                ).subquery()
            )
        )

        return {
            "total": total or 0,
            "due": due or 0,
            "mastered": mastered or 0,
            "reviewed_today": reviewed_today or 0,
        }

    @staticmethod
    async def generate_from_note(
        db: AsyncSession,
        user_id: int,
        note_id: int
    ) -> List[Dict]:
        """
        从笔记内容自动生成闪卡（AI 提取问答对）

        使用现有 AIService 调用 AI 从笔记内容提取问答对
        """
        note = await db.get(LearningNote, note_id)
        if not note or note.user_id != user_id:
            raise NotFoundError(message="笔记不存在")

        # 检查是否已生成过
        existing = await db.execute(
            select(Flashcard).where(
                Flashcard.note_id == note_id,
                Flashcard.source == "auto",
            )
        )
        if existing.scalar_one_or_none():
            return {"message": "该笔记已生成过闪卡", "cards": []}

        # 调用 AI 生成问答对
        from app.services.client.ai_service import AIService
        from app.prompts.learning_note import FLASHCARD_GENERATE_PROMPT

        content = note.content[:3000]
        messages = [
            {"role": "system", "content": FLASHCARD_GENERATE_PROMPT.format(
                title=note.title, content=content
            )}
        ]

        try:
            result = await AIService._chat(messages, temperature=0.3)
            parsed = AIService._extract_json(result)
            qa_pairs = parsed.get("cards", [])

            if not qa_pairs:
                return {"message": "未生成闪卡", "cards": []}

            cards = []
            for qa in qa_pairs[:10]:  # 最多 10 张
                card = Flashcard(
                    user_id=user_id,
                    note_id=note_id,
                    question=qa.get("question", ""),
                    answer=qa.get("answer", ""),
                    source="auto",
                )
                db.add(card)
                cards.append(card)

            await db.commit()
            return {
                "message": f"已生成 {len(cards)} 张闪卡",
                "cards": [{"id": c.id, "question": c.question} for c in cards],
            }
        except Exception as e:
            logger.error(f"闪卡生成失败: {e}")
            return {"message": f"生成失败: {str(e)}", "cards": []}


flashcard_service = FlashcardService()
```

**文件：`app/api/client/v1/flashcard.py`**

```python
from fastapi import APIRouter, Depends, Query
from pydantic import BaseModel
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.client.deps import get_current_user
from app.db.session import get_db
from app.models.user import User
from app.schemas.response import ApiResponse

router = APIRouter()


class ReviewBody(BaseModel):
    quality: int  # 0-5


@router.get("/due")
async def get_due_cards(
    limit: int = Query(20, ge=1, le=50),
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """获取待复习的闪卡"""
    from app.services.client.flashcard_service import flashcard_service
    cards = await flashcard_service.get_due_cards(
        db=db, user_id=current_user.id, limit=limit
    )
    return ApiResponse.success(data={"cards": cards, "count": len(cards)})


@router.post("/{card_id}/review")
async def review_card(
    card_id: int,
    body: ReviewBody,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """复习闪卡并评分"""
    from app.services.client.flashcard_service import flashcard_service
    result = await flashcard_service.review_card(
        db=db, user_id=current_user.id, card_id=card_id,
        quality=body.quality,
    )
    return ApiResponse.success(data=result)


@router.get("/stats")
async def get_flashcard_stats(
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """获取闪卡学习统计"""
    from app.services.client.flashcard_service import flashcard_service
    stats = await flashcard_service.get_stats(db=db, user_id=current_user.id)
    return ApiResponse.success(data=stats)


@router.post("/generate/{note_id}")
async def generate_cards(
    note_id: int,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
):
    """从笔记生成闪卡"""
    from app.services.client.flashcard_service import flashcard_service
    result = await flashcard_service.generate_from_note(
        db=db, user_id=current_user.id, note_id=note_id
    )
    return ApiResponse.success(data=result)
```

import { ref, computed, onMounted } from 'vue'
import { getDueCards, reviewCard, getFlashcardStats } from '../api/learningNote'

const cards = ref([])
const currentIndex = ref(0)
const showAnswer = ref(false)
const stats = ref({})

const currentCard = computed(() => cards.value[currentIndex.value] || {})

async function loadData() {
  const [cardData, statsData] = await Promise.all([
    getDueCards(),
    getFlashcardStats(),
  ])
  cards.value = cardData.cards || []
  stats.value = statsData
}

async function review(quality) {
  try {
    await reviewCard(currentCard.value.id, quality)
    // 下一张
    if (currentIndex.value < cards.value.length - 1) {
      currentIndex.value++
      showAnswer.value = false
    } else {
      // 复习完毕
      await loadData()
    }
  } catch (e) {
    console.error(e)
  }
}
